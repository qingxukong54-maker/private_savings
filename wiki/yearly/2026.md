## 📊 年度概览

```dataviewjs
(async () => {
  const container = this.container;
  const sourcePath = "raw/transactions/transactions/index.md";

  try {
    const file = app.vault.getAbstractFileByPath(sourcePath);
    if (!file) {
      container.innerHTML = '<p style="color:#888;">⚠️ 找不到数据文件</p>';
      return;
    }

    const text = await app.vault.read(file);
    const blocks = text.match(/```yaml\n([\s\S]*?)```/g) || [];

    const records = [];
    blocks.forEach(block => {
      const yaml = block.replace(/```yaml\n?/, '').replace(/```\n?/g, '');
      const record = {};
      yaml.split('\n').forEach(line => {
        const colonIdx = line.indexOf(':');
        if (colonIdx > 0) {
          const key = line.substring(0, colonIdx).trim();
          let val = line.substring(colonIdx + 1).trim();
          if (val === 'null' || val === '') val = null;
          else if (!isNaN(val) && val !== '') val = parseFloat(val);
          else if (val === 'true') val = true;
          else if (val === 'false') val = false;
          record[key] = val;
        }
      });
      if (record.id) records.push(record);
    });

    // 年度统计
    const yearRecords = records.filter(r => r.date && r.date.startsWith('2026'));
    let totalIncome = 0, totalExpense = 0;
    const byMonth = {};
    const byCategory = {};

    yearRecords.forEach(r => {
      const amt = Math.abs(r.amount || 0);
      const month = r.date.substring(0, 7);
      if (r.type === 'income') totalIncome += amt;
      else if (r.type === 'expense') {
        totalExpense += amt;
        byCategory[r.category] = (byCategory[r.category] || 0) + amt;
      }
      if (!byMonth[month]) byMonth[month] = { income: 0, expense: 0 };
      if (r.type === 'income') byMonth[month].income += amt;
      else if (r.type === 'expense') byMonth[month].expense += amt;
    });

    const netSavings = totalIncome - totalExpense;

    // HTML
    let html = '<div style="display:grid;grid-template-columns:repeat(3,1fr);gap:12px;margin:16px 0;">';
    html += `<div style="background:#e8f5e9;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#2e7d32;">年度收入</div>
      <div style="font-size:24px;font-weight:bold;color:#2e7d32;">¥${totalIncome.toFixed(2)}</div>
    </div>`;
    html += `<div style="background:#ffebee;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#c62828;">年度支出</div>
      <div style="font-size:24px;font-weight:bold;color:#c62828;">¥${totalExpense.toFixed(2)}</div>
    </div>`;
    html += `<div style="background:#e3f2fd;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#1565c0;">年度结余</div>
      <div style="font-size:24px;font-weight:bold;color:#1565c0;">¥${netSavings.toFixed(2)}</div>
    </div>`;
    html += '</div>';

    container.innerHTML = html;
  } catch (err) {
    container.innerHTML = `<p style="color:red;">⚠️ 错误: ${err.message}</p>`;
  }
})();
```

## 📂 分类汇总

```dataviewjs
(async () => {
  const container = this.container;
  const sourcePath = "raw/transactions/transactions/index.md";

  try {
    const file = app.vault.getAbstractFileByPath(sourcePath);
    if (!file) {
      container.innerHTML = '<p style="color:#888;">⚠️ 找不到数据文件</p>';
      return;
    }

    const text = await app.vault.read(file);
    const blocks = text.match(/```yaml\n([\s\S]*?)```/g) || [];

    const records = [];
    blocks.forEach(block => {
      const yaml = block.replace(/```yaml\n?/, '').replace(/```\n?/g, '');
      const record = {};
      yaml.split('\n').forEach(line => {
        const colonIdx = line.indexOf(':');
        if (colonIdx > 0) {
          const key = line.substring(0, colonIdx).trim();
          let val = line.substring(colonIdx + 1).trim();
          if (val === 'null' || val === '') val = null;
          else if (!isNaN(val) && val !== '') val = parseFloat(val);
          else if (val === 'true') val = true;
          else if (val === 'false') val = false;
          record[key] = val;
        }
      });
      if (record.id) records.push(record);
    });

    const yearRecords = records.filter(r => r.date && r.date.startsWith('2026') && r.type === 'expense');
    const byCategory = {};
    let total = 0;
    yearRecords.forEach(r => {
      const amt = Math.abs(r.amount || 0);
      total += amt;
      byCategory[r.category] = (byCategory[r.category] || 0) + amt;
    });

    let html = '<table style="width:100%;border-collapse:collapse;font-size:14px;">';
    html += '<tr style="background:#f5f5f5;">';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">分类</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:right;">金额</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:right;">占比</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;">分布</th>';
    html += '</tr>';

    Object.entries(byCategory).sort((a,b) => b[1]-a[1]).forEach(([cat, amt], i) => {
      const pct = (amt/total*100).toFixed(1);
      const barLen = Math.round(pct / 5);
      const bar = '█'.repeat(barLen) + '░'.repeat(20 - barLen);
      const bg = i % 2 === 0 ? '#fff' : '#fafafa';
      html += `<tr style="background:${bg};">
        <td style="padding:6px 8px;border:1px solid #ddd;font-weight:500;">${cat}</td>
        <td style="padding:6px 8px;border:1px solid #ddd;text-align:right;">¥${amt.toFixed(2)}</td>
        <td style="padding:6px 8px;border:1px solid #ddd;text-align:right;">${pct}%</td>
        <td style="padding:6px 8px;border:1px solid #ddd;font-family:monospace;color:#1565c0;">${bar}</td>
      </tr>`;
    });

    html += '</table>';
    container.innerHTML = html;
  } catch (err) {
    container.innerHTML = `<p style="color:red;">⚠️ 错误: ${err.message}</p>`;
  }
})();
```
