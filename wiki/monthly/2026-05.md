```dataviewjs
// 月份筛选（自动获取当前日期）
const TARGET_MONTH = "2026-05";

(async () => {
  const container = this.container;
  const sourcePath = "raw/transactions/transactions/index.md";

  try {
    const file = app.vault.getAbstractFileByPath(sourcePath);
    if (!file) {
      container.innerHTML = '<p style="color:#888;">⚠️ 找不到数据文件</p>';
      return;
    }

    const text = await app.vault.read(file);
    const blocks = text.match(/```yaml\n([\s\S]*?)```/g) || [];

    // 解析 YAML 代码块
    const records = [];
    blocks.forEach(block => {
      const yaml = block.replace(/```yaml\n?/, '').replace(/```\n?/g, '');
      const record = {};
      yaml.split('\n').forEach(line => {
        const colonIdx = line.indexOf(':');
        if (colonIdx > 0) {
          const key = line.substring(0, colonIdx).trim();
          let val = line.substring(colonIdx + 1).trim();
          if (val === 'null' || val === '') val = null;
          else if (!isNaN(val) && val !== '') val = parseFloat(val);
          else if (val === 'true') val = true;
          else if (val === 'false') val = false;
          record[key] = val;
        }
      });
      if (record.id) records.push(record);
    });

    // 月份筛选
    const filtered = records.filter(r => {
      if (!r.date) return false;
      const month = r.date.substring(0, 7);
      return !TARGET_MONTH || month === TARGET_MONTH;
    });

    // 统计
    let totalIncome = 0, totalExpense = 0;
    const byCategory = {};
    const byAccount = {};

    filtered.forEach(r => {
      const amt = Math.abs(r.amount || 0);
      if (r.type === 'income') totalIncome += amt;
      else if (r.type === 'expense') {
        totalExpense += amt;
        byCategory[r.category] = (byCategory[r.category] || 0) + amt;
      }
      if (r.account) byAccount[r.account] = (byAccount[r.account] || 0) + amt;
    });

    const netSavings = totalIncome - totalExpense;

    // HTML
    let html = '<div style="display:grid;grid-template-columns:repeat(3,1fr);gap:12px;margin:16px 0;">';
    html += `<div style="background:#e8f5e9;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#2e7d32;">收入</div>
      <div style="font-size:24px;font-weight:bold;color:#2e7d32;">¥${totalIncome.toFixed(2)}</div>
    </div>`;
    html += `<div style="background:#ffebee;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#c62828;">支出</div>
      <div style="font-size:24px;font-weight:bold;color:#c62828;">¥${totalExpense.toFixed(2)}</div>
    </div>`;
    html += `<div style="background:#e3f2fd;padding:16px;border-radius:8px;text-align:center;">
      <div style="font-size:12px;color:#1565c0;">结余</div>
      <div style="font-size:24px;font-weight:bold;color:#1565c0;">¥${netSavings.toFixed(2)}</div>
    </div>`;
    html += '</div>';

    // 分类明细
    if (Object.keys(byCategory).length > 0) {
      html += '<details style="margin-top:12px;"><summary style="cursor:pointer;font-weight:bold;">📂 分类明细</summary>';
      html += '<table style="width:100%;margin-top:8px;border-collapse:collapse;">';
      html += '<tr style="background:#f5f5f5;"><th style="padding:8px;border:1px solid #ddd;text-align:left;">分类</th><th style="padding:8px;border:1px solid #ddd;text-align:right;">金额</th><th style="padding:8px;border:1px solid #ddd;text-align:right;">占比</th></tr>';
      Object.entries(byCategory).sort((a,b) => b[1]-a[1]).forEach(([cat, amt]) => {
        html += `<tr><td style="padding:6px 8px;border:1px solid #ddd;">${cat}</td><td style="padding:6px 8px;border:1px solid #ddd;text-align:right;">¥${amt.toFixed(2)}</td><td style="padding:6px 8px;border:1px solid #ddd;text-align:right;">${(amt/totalExpense*100).toFixed(1)}%</td></tr>`;
      });
      html += '</table></details>';
    }

    container.innerHTML = html;
  } catch (err) {
    container.innerHTML = `<p style="color:red;">⚠️ 错误: ${err.message}</p>`;
  }
})();
```

```dataviewjs
// 月份筛选
const TARGET_MONTH = "2026-05";

(async () => {
  const container = this.container;
  const sourcePath = "raw/transactions/transactions/index.md";

  try {
    const file = app.vault.getAbstractFileByPath(sourcePath);
    if (!file) {
      container.innerHTML = '<p style="color:#888;">⚠️ 找不到数据文件</p>';
      return;
    }

    const text = await app.vault.read(file);
    const blocks = text.match(/```yaml\n([\s\S]*?)```/g) || [];

    const records = [];
    blocks.forEach(block => {
      const yaml = block.replace(/```yaml\n?/, '').replace(/```\n?/g, '');
      const record = {};
      yaml.split('\n').forEach(line => {
        const colonIdx = line.indexOf(':');
        if (colonIdx > 0) {
          const key = line.substring(0, colonIdx).trim();
          let val = line.substring(colonIdx + 1).trim();
          if (val === 'null' || val === '') val = null;
          else if (!isNaN(val) && val !== '') val = parseFloat(val);
          else if (val === 'true') val = true;
          else if (val === 'false') val = false;
          record[key] = val;
        }
      });
      if (record.id) records.push(record);
    });

    // 筛选本月
    const filtered = records.filter(r => {
      if (!r.date) return false;
      const month = r.date.substring(0, 7);
      return !TARGET_MONTH || month === TARGET_MONTH;
    }).sort((a,b) => (b.date || '').localeCompare(a.date || ''));

    if (filtered.length === 0) {
      container.innerHTML = '<p style="color:#888;">暂无交易记录</p>';
      return;
    }

    // 表格
    let html = '<table style="width:100%;border-collapse:collapse;font-size:14px;">';
    html += '<tr style="background:#f5f5f5;">';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">日期</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">类型</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">分类</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">描述</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:left;">账户</th>';
    html += '<th style="padding:8px;border:1px solid #ddd;text-align:right;">金额</th>';
    html += '</tr>';

    filtered.forEach(r => {
      const isIncome = r.type === 'income';
      const color = isIncome ? '#2e7d32' : '#c62828';
      const prefix = isIncome ? '+' : '-';
      const bg = filtered.indexOf(r) % 2 === 0 ? '#fff' : '#fafafa';

      html += `<tr style="background:${bg};">`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;">${r.date || '-'}</td>`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;"><span style="color:${color};">${isIncome ? '收入' : '支出'}</span></td>`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;">${r.category || '-'}</td>`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;">${r.note || '-'}</td>`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;">${r.account || '-'}</td>`;
      html += `<td style="padding:6px 8px;border:1px solid #ddd;text-align:right;color:${color};font-weight:bold;">${prefix}¥${Math.abs(r.amount || 0).toFixed(2)}</td>`;
      html += '</tr>';
    });

    html += '</table>';
    container.innerHTML = html;
  } catch (err) {
    container.innerHTML = `<p style="color:red;">⚠️ 错误: ${err.message}</p>`;
  }
})();
```
